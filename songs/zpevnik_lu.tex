% Příkaz, který nafituje obrázek na zbývající místo na stránce:
\newlength{\imgwidth}
\newlength{\imgheight}
\newcommand{\scalegraphics}[1]{						
    \settowidth{\imgwidth}{\includegraphics{#1}}
    \settoheight{\imgheight}{\includegraphics{#1}}
    \setlength{\imgwidth}{\minof{\imgwidth}{\textwidth}}
    \setlength{\imgheight}{\minof{\imgheight}{\the\dimexpr\pagegoal-\pagetotal\relax}}
    \includegraphics[width=\imgwidth, height=\imgheight, keepaspectratio=true]{#1}
}
